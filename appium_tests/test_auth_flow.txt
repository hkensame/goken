"""
Appium UI 测试 —— LoginScreen & RegisterScreen
================================================
测试目标页面：
  - LoginScreen  (CampusTradeScreens.kt)
  - RegisterScreen (CampusTradeScreens.kt)

运行前提：
  1. 启动 Appium Server：  appium --port 4723
  2. 已连接真机或启动模拟器（adb devices 能看到设备）
  3. 已在设备上安装 debug APK：
       ./gradlew installDebug
  4. 安装 Python 依赖：
       pip install Appium-Python-Client pytest

运行命令：
  pytest appium_tests/test_auth_flow.py -v
"""

import time
import pytest
from appium import webdriver
from appium.options import UiAutomator2Options
from appium.webdriver.common.appiumby import AppiumBy
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import TimeoutException


# ──────────────────────────────────────────────
# 配置区（按实际情况修改）
# ──────────────────────────────────────────────
APPIUM_URL = "http://127.0.0.1:4723"

CAPS = {
    "platformName": "Android",
    "appium:automationName": "UiAutomator2",
    "appium:appPackage": "com.example.myapplication",
    "appium:appActivity": "com.example.myapplication.MainActivity",
    # 不重装 APK，直接启动已安装的 App
    "appium:noReset": True,
    "appium:fullReset": False,
    # 等待 App 启动超时（毫秒）
    "appium:newCommandTimeout": 120,
    "appium:autoGrantPermissions": True,
    # 将设备的 UDID 填在这里（adb devices 看到的值），留空表示使用第一台设备
    # "appium:udid": "emulator-5554",
}

DEFAULT_WAIT = 10  # 元素等待超时（秒）


# ──────────────────────────────────────────────
# Fixture：每个 Test 独立启动 / 关闭 Session
# ──────────────────────────────────────────────
@pytest.fixture
def driver():
    options = UiAutomator2Options().load_capabilities(CAPS)
    _driver = webdriver.Remote(APPIUM_URL, options=options)
    _driver.implicitly_wait(DEFAULT_WAIT)
    yield _driver
    _driver.quit()


# ──────────────────────────────────────────────
# 工具函数
# ──────────────────────────────────────────────
def wait_for_text(driver, text: str, timeout: int = DEFAULT_WAIT):
    """等待含指定文本的节点出现并返回该元素。"""
    return WebDriverWait(driver, timeout).until(
        EC.presence_of_element_located(
            (AppiumBy.ANDROID_UIAUTOMATOR,
             f'new UiSelector().text("{text}")')
        )
    )


def find_by_text(driver, text: str):
    return driver.find_element(
        AppiumBy.ANDROID_UIAUTOMATOR,
        f'new UiSelector().text("{text}")'
    )


def find_input_by_hint(driver, hint: str):
    """通过 hint（label）定位 OutlinedTextField 输入框。"""
    return driver.find_element(
        AppiumBy.ANDROID_UIAUTOMATOR,
        f'new UiSelector().description("{hint}")'
    )


def find_all_edit_texts(driver):
    return driver.find_elements(AppiumBy.CLASS_NAME, "android.widget.EditText")


def unique_student_id() -> str:
    return f"stu_{int(time.time() * 1000) % 100_000_000}"


# ──────────────────────────────────────────────
# 测试：LoginScreen
# ──────────────────────────────────────────────
class TestLoginScreen:

    def test_login_screen_shows_initial_elements(self, driver):
        """
        TC-L-01：冷启动后 LoginScreen 应显示：
          - 标题「登录」
          - 学号输入框
          - 密码输入框
          - 登录按钮
          - 「没有账号？去注册」文本按钮
        """
        wait_for_text(driver, "登录")

        assert find_by_text(driver, "登录").is_displayed()
        assert find_by_text(driver, "没有账号？去注册").is_displayed()

        fields = find_all_edit_texts(driver)
        assert len(fields) >= 2, "应至少有学号、密码两个输入框"

    def test_login_empty_fields_shows_error(self, driver):
        """
        TC-L-02：不填任何内容直接点击登录，应显示错误提示（非空校验）。
        Compose 中错误消息由 err 状态控制，颜色为 error，文本非空即可。
        """
        wait_for_text(driver, "登录")

        # 直接点击登录按钮（不填内容）
        login_btn = driver.find_element(
            AppiumBy.ANDROID_UIAUTOMATOR,
            'new UiSelector().text("登录").className("android.widget.Button")'
        )
        login_btn.click()

        # 等待错误文本出现（任意非空文本的 error 颜色节点）
        time.sleep(1)
        # Compose 错误文本通过 Text 组件渲染，内容取决于业务逻辑
        # 这里验证页面依然在 LoginScreen（没有跳转走）
        assert find_by_text(driver, "没有账号？去注册").is_displayed(), \
            "登录失败后应留在登录页"

    def test_login_wrong_password_shows_error(self, driver):
        """
        TC-L-03：使用不存在的学号 + 随机密码登录，应在页面上显示错误消息。
        """
        wait_for_text(driver, "登录")

        fields = find_all_edit_texts(driver)
        fields[0].clear()
        fields[0].send_keys("nonexistent_user_99999")
        fields[1].clear()
        fields[1].send_keys("wrong_password_xyz")

        login_btn = driver.find_element(
            AppiumBy.ANDROID_UIAUTOMATOR,
            'new UiSelector().text("登录").className("android.widget.Button")'
        )
        login_btn.click()

        # 等待错误提示出现（最多 8 秒）
        time.sleep(2)
        # 验证没有跳转到 MainShell（「校园二手」页头不存在）
        try:
            driver.find_element(
                AppiumBy.ANDROID_UIAUTOMATOR,
                'new UiSelector().text("校园二手")'
            )
            pytest.fail("错误密码不应进入主页面")
        except Exception:
            pass  # 符合预期：未跳转

        # 登录页仍然存在
        assert find_by_text(driver, "没有账号？去注册").is_displayed()

    def test_login_navigate_to_register(self, driver):
        """
        TC-L-04：点击「没有账号？去注册」应跳转到注册页。
        """
        wait_for_text(driver, "登录")

        find_by_text(driver, "没有账号？去注册").click()

        # 等待注册页加载
        wait_for_text(driver, "注册")
        assert find_by_text(driver, "注册并登录").is_displayed()
        assert find_by_text(driver, "昵称（可选）").is_displayed() or \
               len(find_all_edit_texts(driver)) >= 3, "注册页应有3个输入框"

# 测试：RegisterScreen
# ──────────────────────────────────────────────
class TestRegisterScreen:

    def _go_to_register(self, driver):
        """从 LoginScreen 导航到 RegisterScreen。"""
        wait_for_text(driver, "登录")
        find_by_text(driver, "没有账号？去注册").click()
        wait_for_text(driver, "注册")

    def test_register_screen_shows_initial_elements(self, driver):
        """
        TC-R-01：进入注册页后应显示：
          - 标题「注册」
          - 3 个输入框（学号、密码、昵称）
          - 「注册并登录」按钮
          - 「返回登录」按钮
        """
        self._go_to_register(driver)

        assert find_by_text(driver, "注册并登录").is_displayed()
        assert find_by_text(driver, "返回登录").is_displayed()

        fields = find_all_edit_texts(driver)
        assert len(fields) >= 3, "注册页应有学号、密码、昵称三个输入框"

    def test_register_back_to_login(self, driver):
        """
        TC-R-02：点击「返回登录」应回到登录页。
        """
        self._go_to_register(driver)

        find_by_text(driver, "返回登录").click()

        wait_for_text(driver, "登录")
        assert find_by_text(driver, "没有账号？去注册").is_displayed()

    def test_register_empty_fields_stays_on_page(self, driver):
        """
        TC-R-03：不填任何内容点击「注册并登录」，应停留在注册页（不跳转）。
        """
        self._go_to_register(driver)

        find_by_text(driver, "注册并登录").click()
        time.sleep(1)

        # 不应跳转到主页
        try:
            driver.find_element(
                AppiumBy.ANDROID_UIAUTOMATOR,
                'new UiSelector().text("校园二手")'
            )
            pytest.fail("空字段不应注册成功并进入主页")
        except Exception:
            pass

        # 仍在注册页
        assert find_by_text(driver, "注册并登录").is_displayed()

    def test_register_success_and_enters_main(self, driver):
        """
        TC-R-04（完整流程）：
          1. 填写唯一学号 + 密码 + 昵称
          2. 点击「注册并登录」
          3. 注册成功后自动登录，应进入 MainShell 首页
             （顶栏显示「校园二手」，底部导航栏显示「首页」「发布」等）
        """
        self._go_to_register(driver)

        sid = unique_student_id()
        fields = find_all_edit_texts(driver)
        fields[0].send_keys(sid)
        fields[1].send_keys("appium_pass_123")
        fields[2].send_keys("Appium测试用户")

        find_by_text(driver, "注册并登录").click()

        # 最多等 15 秒进入首页
        try:
            wait_for_text(driver, "校园二手", timeout=15)
        except TimeoutException:
            pytest.fail("注册并登录后未能进入主页面，超时")

        assert find_by_text(driver, "校园二手").is_displayed()
        assert find_by_text(driver, "首页").is_displayed()
        assert find_by_text(driver, "发布").is_displayed()

    def test_register_duplicate_student_id_shows_error(self, driver):
        """
        TC-R-05：注册已存在的学号应显示错误提示，且停留在注册页。

        前置条件：TC-R-04 已执行过，或手动预先注册同一学号。
        本测试用固定学号 'appium_dup_test' 做第二次注册验证重复逻辑。
        """
        self._go_to_register(driver)

        fields = find_all_edit_texts(driver)
        fields[0].send_keys("appium_dup_test")
        fields[1].send_keys("any_password")
        # 昵称留空

        find_by_text(driver, "注册并登录").click()

        time.sleep(2)

        # 若已存在该学号：停留在注册页 + 出现错误提示
        # 若不存在（首次）：注册成功跳转主页 —— 此时跳过断言
        still_on_register = True
        try:
            find_by_text(driver, "注册并登录")
        except Exception:
            still_on_register = False

        if still_on_register:
            # 验证有错误文本存在（颜色无法直接断言，确保页面有文字反馈即可）
            page_source = driver.page_source
            assert any(kw in page_source for kw in ["错误", "失败", "已存在", "error"]), \
                "重复学号应有错误提示"
